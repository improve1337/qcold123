local function createGlow(color, position, size)
    local glow = Instance.new("Frame")
    glow.AnchorPoint = Vector2.new(0.5, 0.5)
    glow.Position = position
    glow.Size = size
    glow.BackgroundColor3 = color
    glow.BackgroundTransparency = 0.45
    glow.BorderSizePixel = 0
    glow.Parent = Background

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(1, 0)
    corner.Parent = glow

    local gradient = Instance.new("UIGradient")
    gradient.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 0.2),
        NumberSequenceKeypoint.new(1, 1)
    })
    gradient.Parent = glow

    TweenService:Create(
        glow,
        TweenInfo.new(4, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
        {
            Size = size + UDim2.fromOffset(100, 100),
            Position = position + UDim2.fromOffset(40, -25)
        }
    ):Play()
end

local function createRing(size, transparency)
    local ring = Instance.new("Frame")
    ring.AnchorPoint = Vector2.new(0.5, 0.5)
    ring.Position = UDim2.fromScale(0.5, 0.5)
    ring.Size = UDim2.fromOffset(size, size)
    ring.BackgroundTransparency = 1
    ring.Parent = Background

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(1, 0)
    corner.Parent = ring

    local stroke = Instance.new("UIStroke")
    stroke.Thickness = 1.5
    stroke.Transparency = transparency
    stroke.Color = Color3.fromRGB(100, 100, 255)
    stroke.Parent = ring

    local gradient = Instance.new("UIGradient")
    gradient.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromRGB(40, 180, 255)),
        ColorSequenceKeypoint.new(0.5, Color3.fromRGB(170, 50, 255)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(40, 180, 255))
    })
    gradient.Parent = stroke

    TweenService:Create(
        gradient,
        TweenInfo.new(5, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1),
        {Rotation = 360}
    ):Play()

    TweenService:Create(
        ring,
        TweenInfo.new(3, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
        {Size = UDim2.fromOffset(size + 25, size + 25)}
    ):Play()
end

local languageButtons = {}

local function createLanguageButton(code, label)
    local button = Instance.new("TextButton")
    button.Size = UDim2.fromOffset(88, 34)
    button.BackgroundColor3 = Color3.fromRGB(30, 30, 48)
    button.BackgroundTransparency = 0.35
    button.Text = label
    button.TextColor3 = Color3.fromRGB(170, 175, 205)
    button.Font = Enum.Font.GothamBold
    button.TextSize = 14
    button.AutoButtonColor = false
    button.Parent = LanguageRow

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 10)
    corner.Parent = button

    local stroke = Instance.new("UIStroke")
    stroke.Thickness = 1.2
    stroke.Color = Color3.fromRGB(90, 110, 200)
    stroke.Transparency = 0.55
    stroke.Parent = button

    button.MouseEnter:Connect(function()
        if currentLanguage ~= code then
            stroke.Transparency = 0.2
        end
    end)

    button.MouseLeave:Connect(function()
        if currentLanguage ~= code then
            stroke.Transparency = 0.55
        end
    end)

    button.MouseButton1Click:Connect(function()
        currentLanguage = code
        setCaserLocale(code)
        Title.Text = texts().title
        Review.Text = texts().review
        Status.Text = texts().status

        for buttonCode, entry in pairs(languageButtons) do
            local active = buttonCode == code
            entry.button.BackgroundColor3 = active and Color3.fromRGB(70, 120, 255) or Color3.fromRGB(30, 30, 48)
            entry.button.BackgroundTransparency = active and 0.15 or 0.35
            entry.button.TextColor3 = active and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(170, 175, 205)
            entry.stroke.Transparency = active and 0 or 0.55
        end
    end)

    languageButtons[code] = { button = button, stroke = stroke }
    return button
end

createLanguageButton("ru", "РУС")
createLanguageButton("en", "EN")

-- Подсветка языка по умолчанию
do
    local entry = languageButtons[currentLanguage]
    if entry then
        entry.button.BackgroundColor3 = Color3.fromRGB(70, 120, 255)
        entry.button.BackgroundTransparency = 0.15
        entry.button.TextColor3 = Color3.fromRGB(255, 255, 255)
        entry.stroke.Transparency = 0
    end
end

-- Запуск CASER
local startTime = os.clock()
local finished = false
local loadError = nil

task.spawn(function()
    local success, err = pcall(function()
        local source = game:HttpGet(SCRIPT_URL)
        assert(type(source) == "string" and #source > 0, "Не удалось загрузить скрипт")

        local compiled, compileError = loadstring(source)
        assert(type(compiled) == "function", compileError or "Ошибка компиляции")

        setCaserLocale(currentLanguage)
        compiled()
    end)

    if not success then
        loadError = err
        warn("[Caser Loading] " .. tostring(err))
    end

    finished = true
end)

-- Не убираем заставку раньше 3.8 секунды
while os.clock() - startTime < 3.8 do
    task.wait(0.1)
end

-- Ждём завершения запуска, но не бесконечно
local waitStart = os.clock()
while not finished and os.clock() - waitStart < 25 do
    task.wait(0.1)
end

-- Плавное исчезновение
for _, obj in ipairs(Background:GetDescendants()) do
    if obj:IsA("TextLabel") then
        TweenService:Create(obj, TweenInfo.new(1.2), {
            TextTransparency = 1
        }):Play()
    elseif obj:IsA("Frame") then
        TweenService:Create(obj, TweenInfo.new(1.2), {
            BackgroundTransparency = 1
        }):Play()
    elseif obj:IsA("UIStroke") then
        TweenService:Create(obj, TweenInfo.new(1.2), {
            Transparency = 1
        }):Play()
    elseif obj:IsA("TextButton") then
        TweenService:Create(obj, TweenInfo.new(1.2), {
            BackgroundTransparency = 1,
            TextTransparency = 1
        }):Play()
    end
end
